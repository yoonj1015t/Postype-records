<script>
    // 페이지 로드 시 데이터 불러오기
    let quotes = JSON.parse(localStorage.getItem('myQuotes')) || [];
    displayQuotes();

    function addQuote() {
        const title = document.getElementById('title').value;
        const author = document.getElementById('author').value;
        const quote = document.getElementById('quote').value;
        const tags = document.getElementById('tags').value;

        if(!title || !quote) return alert("제목과 대사는 필수입니다!");

        const newQuote = { id: Date.now(), title, author, quote, tags };
        quotes.push(newQuote);
        localStorage.setItem('myQuotes', JSON.stringify(quotes));
        
        // 입력창 비우기
        document.querySelectorAll('input, textarea').forEach(input => input.value = '');
        displayQuotes();
    }

    function displayQuotes() {
        const searchTerm = document.getElementById('search').value.toLowerCase();
        const list = document.getElementById('quoteList');
        list.innerHTML = '';

        const filtered = quotes.filter(q => 
            q.title.toLowerCase().includes(searchTerm) || 
            q.author.toLowerCase().includes(searchTerm) || 
            q.tags.toLowerCase().includes(searchTerm) ||
            q.quote.toLowerCase().includes(searchTerm)
        );

        filtered.reverse().forEach(q => {
            list.innerHTML += `
                <div class="quote-card">
                    <button class="delete-btn" onclick="deleteQuote(${q.id})">✕</button>
                    <div class="quote-text">${q.quote}</div>
                    <div class="quote-info"><strong>${q.title}</strong> | ${q.author}</div>
                    <div class="tag">#${q.tags}</div>
                </div>
            `;
        });
    }

    function deleteQuote(id) {
        if(confirm("이 대사를 삭제할까요?")) {
            quotes = quotes.filter(q => q.id !== id);
            localStorage.setItem('myQuotes', JSON.stringify(quotes));
            displayQuotes();
        }
    }
</script>
